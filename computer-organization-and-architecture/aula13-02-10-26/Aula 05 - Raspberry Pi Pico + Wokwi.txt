from machine import Pin, SPI
import time


# =========================================================
# SPI
# =========================================================

spi = SPI(
    1,
    baudrate=1000000,
    polarity=0,
    phase=0,
    sck=Pin(10),
    mosi=Pin(11)
)

cs = Pin(9, Pin.OUT)
cs.value(1)


# =========================================================
# FUNÇÃO PARA ENVIAR UM COMANDO AO MAX7219
# =========================================================

def enviar(registro, valor):

    cs.value(0)

    spi.write(bytes([registro, valor]))

    cs.value(1)


# =========================================================
# INICIALIZAÇÃO DO MAX7219
# =========================================================

# Desliga o modo shutdown
enviar(0x0C, 0x01)

# Não utilizar decodificação BCD
enviar(0x09, 0x00)

# Utilizar todas as 8 linhas
enviar(0x0B, 0x07)

# Intensidade
enviar(0x0A, 0x08)


# =========================================================
# LETRA A
# =========================================================

imagem = [
    0x18,
    0x3C,
    0x66,
    0x66,
    0x7E,
    0x66,
    0x66,
    0x00
]


# =========================================================
# ENVIAR OS 8 BYTES PARA A MATRIZ
# =========================================================

for linha in range(8):

    enviar(linha + 1, imagem[linha])


# =========================================================
# MANTER A SIMULAÇÃO FUNCIONANDO
# =========================================================

while True:

    time.sleep(1)
